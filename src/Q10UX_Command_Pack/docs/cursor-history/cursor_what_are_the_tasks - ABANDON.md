export const formatRelativeTime = (date: Date | string): string => {
  const dateObj = typeof date === 'string' ? new Date(date) : date;
  const now = new Date();
  const diffInMs = now.getTime() - dateObj.getTime();
  const diffInDays = Math.floor(diffInMs / (1000 * 60 * 60 * 24));
  
  if (diffInDays === 0) return 'Today';
  if (diffInDays === 1) return 'Yesterday';
  if (diffInDays < 7) return `${diffInDays} days ago`;
  if (diffInDays < 30) return `${Math.floor(diffInDays / 7)} weeks ago`;
  if (diffInDays < 365) return `${Math.floor(diffInDays / 30)} months ago`;
  
  return `${Math.floor(diffInDays / 365)} years ago`;
};

export const formatDuration = (startDate: Date | string, endDate?: Date | string): string => {
  const start = typeof startDate === 'string' ? new Date(startDate) : startDate;
  const end = endDate ? (typeof endDate === 'string' ? new Date(endDate) : endDate) : new Date();
  
  const diffInMs = end.getTime() - start.getTime();
  const diffInDays = Math.floor(diffInMs / (1000 * 60 * 60 * 24));
  const diffInMonths = Math.floor(diffInDays / 30);
  const diffInYears = Math.floor(diffInDays / 365);
  
  if (diffInYears > 0) {
    return `${diffInYears} year${diffInYears > 1 ? 's' : ''}`;
  }
  if (diffInMonths > 0) {
    return `${diffInMonths} month${diffInMonths > 1 ? 's' : ''}`;
  }
  if (diffInDays > 0) {
    return `${diffInDays} day${diffInDays > 1 ? 's' : ''}`;
  }
  
  return 'Less than a day';

/**
 * Custom hook for managing localStorage with TypeScript support
 */
export function useLocalStorage<T>(
  key: string,
  initialValue: T
): [T, (value: T | ((val: T) => T)) => void] {
  // Get from local storage then parse stored json or return initialValue
  const [storedValue, setStoredValue] = useState<T>(() => {
    if (typeof window === 'undefined') {
      return initialValue;
    }
    
    try {
      const item = window.localStorage.getItem(key);
      return item ? JSON.parse(item) : initialValue;
    } catch (error) {
      console.error(`Error reading localStorage key "${key}":`, error);
      return initialValue;
    }
  });

  // Return a wrapped version of useState's setter function that persists the new value to localStorage
  const setValue = (value: T | ((val: T) => T)) => {
    try {
      // Allow value to be a function so we have the same API as useState
      const valueToStore = value instanceof Function ? value(storedValue) : value;
      setStoredValue(valueToStore);
      
      // Save to local storage
      if (typeof window !== 'undefined') {
        window.localStorage.setItem(key, JSON.stringify(valueToStore));
      }
    } catch (error) {
      console.error(`Error setting localStorage key "${key}":`, error);
    }
  };

  return [storedValue, setValue];
}

import React, { createContext, useContext, useEffect, useState } from 'react';

type Theme = 'light' | 'dark';

interface ThemeContextType {
  theme: Theme;
  toggleTheme: () => void;
  setTheme: (theme: Theme) => void;
}

const ThemeContext = createContext<ThemeContextType | undefined>(undefined);

export const useTheme = () => {
  const context = useContext(ThemeContext);
  if (context === undefined) {
    throw new Error('useTheme must be used within a ThemeProvider');
  }
  return context;
};

interface ThemeProviderProps {
  children: React.ReactNode;
}

export const ThemeProvider: React.FC<ThemeProviderProps> = ({ children }) => {
  const [theme, setThemeState] = useState<Theme>(() => {
    // Check localStorage first
    const savedTheme = localStorage.getItem('theme') as Theme;
    if (savedTheme) {
      return savedTheme;
    }
    
    // Check system preference
    if (window.matchMedia('(prefers-color-scheme: dark)').matches) {
      return 'dark';
    }
    
    return 'light';
  });

  const setTheme = (newTheme: Theme) => {
    setThemeState(newTheme);
    localStorage.setItem('theme', newTheme);
  };

  const toggleTheme = () => {
    setTheme(theme === 'light' ? 'dark' : 'light');
  };

  useEffect(() => {
    const root = window.document.documentElement;
    
    // Remove existing theme classes
    root.classList.remove('light', 'dark');
    
    // Add current theme class
    root.classList.add(theme);
    
    // Update meta theme-color
    const metaThemeColor = document.querySelector('meta[name="theme-color"]');
    if (metaThemeColor) {
      metaThemeColor.setAttribute(
        'content',
        theme === 'dark' ? '#0a0a0a' : '#fafafa'
      );
    }
  }, [theme]);

  // Listen for system theme changes
  useEffect(() => {
    const mediaQuery = window.matchMedia('(prefers-color-scheme: dark)');
    
    const handleChange = (e: MediaQueryListEvent) => {
      if (!localStorage.getItem('theme')) {
        setTheme(e.matches ? 'dark' : 'light');
      }
    };

    mediaQuery.addEventListener('change', handleChange);
    return () => mediaQuery.removeEventListener('change', handleChange);
  }, []);

  const value = {
    theme,
    toggleTheme,
    setTheme,
  };

  return (
    <ThemeContext.Provider value={value}>
      {children}
    </ThemeContext.Provider>
  );
};

        label,
        isActive: isLast,

#!/usr/bin/env node

import sharp from 'sharp';
import fs from 'fs/promises';
import path from 'path';
import { fileURLToPath } from 'url';

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

// Configuration
const config = {
  inputDir: path.join(__dirname, '../src/assets/images/raw'),
  outputDir: path.join(__dirname, '../src/assets/images'),
  formats: ['webp', 'jpeg', 'png'],
  sizes: {
    thumbnail: { width: 300, height: 300 },
    small: { width: 600, height: 600 },
    medium: { width: 1200, height: 1200 },
    large: { width: 1920, height: 1920 },
    hero: { width: 2560, height: 1440 }
  },
  quality: {
    webp: 80,
    jpeg: 85,
    png: 90
  }
};

// Image processing class
class ImageProcessor {
  constructor() {
    this.processedImages = [];
  }

  async ensureDirectories() {
    const dirs = [
      config.outputDir,
      path.join(config.outputDir, 'thumbnails'),
      path.join(config.outputDir, 'small'),
      path.join(config.outputDir, 'medium'),
      path.join(config.outputDir, 'large'),
      path.join(config.outputDir, 'hero'),
      path.join(config.outputDir, 'webp'),
      path.join(config.outputDir, 'jpeg'),
      path.join(config.outputDir, 'png')
    ];

    for (const dir of dirs) {
      await fs.mkdir(dir, { recursive: true });
    }
  }

  async getImageFiles() {
    try {
      const files = await fs.readdir(config.inputDir);
      return files.filter(file => 
        /\.(jpg|jpeg|png|gif|bmp|tiff|webp)$/i.test(file)
      );
    } catch (error) {
      console.log('No raw images directory found. Creating...');
      await fs.mkdir(config.inputDir, { recursive: true });
      return [];
    }
  }

  generateAltText(filename) {
    // Remove extension and convert to readable text
    const name = path.basename(filename, path.extname(filename));
    return name
      .replace(/[-_]/g, ' ')
      .replace(/\b\w/g, l => l.toUpperCase())
      .trim();
  }

  async processImage(inputPath, filename) {
    const baseName = path.basename(filename, path.extname(filename));
    const altText = this.generateAltText(filename);
    
    console.log(`Processing: ${filename}`);
    
    try {
      const image = sharp(inputPath);
      const metadata = await image.metadata();
      
      const imageData = {
        id: baseName,
        originalName: filename,
        alt: altText,
        width: metadata.width,
        height: metadata.height,
        format: metadata.format,
        sizes: {},
        formats: {}
      };

      // Process different sizes
      for (const [sizeName, size] of Object.entries(config.sizes)) {
        const resized = image.resize(size.width, size.height, {
          fit: 'inside',
          withoutEnlargement: true
        });

        // Process different formats
        for (const format of config.formats) {
          const outputPath = path.join(
            config.outputDir,
            format,
            `${baseName}-${sizeName}.${format}`
          );

          let processed;
          switch (format) {
            case 'webp':
              processed = resized.webp({ quality: config.quality.webp });
              break;
            case 'jpeg':
              processed = resized.jpeg({ quality: config.quality.jpeg });
              break;
            case 'png':
              processed = resized.png({ quality: config.quality.png });
              break;
          }

          await processed.toFile(outputPath);
          
          // Get file size
          const stats = await fs.stat(outputPath);
          
          if (!imageData.sizes[sizeName]) {
            imageData.sizes[sizeName] = {};
          }
          if (!imageData.formats[format]) {
            imageData.formats[format] = {};
          }
          
          imageData.sizes[sizeName][format] = {
            url: `/src/assets/images/${format}/${baseName}-${sizeName}.${format}`,
            width: size.width,
            height: size.height,
            size: stats.size
          };
        }
      }

      this.processedImages.push(imageData);
      console.log(`✅ Processed: ${filename}`);
      
    } catch (error) {
      console.error(`❌ Error processing ${filename}:`, error.message);
    }
  }

  async generateManifest() {
    const manifest = {
      generated: new Date().toISOString(),
      totalImages: this.processedImages.length,
      images: this.processedImages
    };

    const manifestPath = path.join(config.outputDir, 'image-manifest.json');
    await fs.writeFile(manifestPath, JSON.stringify(manifest, null, 2));
    console.log(`📄 Generated manifest: ${manifestPath}`);
  }

  async generateTypeScriptTypes() {
    const types = `// Auto-generated image types
export interface ProcessedImage {
  id: string;
  originalName: string;
  alt: string;
  width: number;
  height: number;
  format: string;
  sizes: {
    [size: string]: {
      [format: string]: {
        url: string;
        width: number;
        height: number;
        size: number;
      };
    };
  };
  formats: {
    [format: string]: {
      [size: string]: {
        url: string;
        width: number;
        height: number;
        size: number;
      };
    };
  };
}

export interface ImageManifest {
  generated: string;
  totalImages: number;
  images: ProcessedImage[];
}

// Available image IDs
export const IMAGE_IDS = ${JSON.stringify(this.processedImages.map(img => img.id), null, 2)} as const;

export type ImageId = typeof IMAGE_IDS[number];
`;

    const typesPath = path.join(config.outputDir, 'image-types.ts');
    await fs.writeFile(typesPath, types);
    console.log(`📝 Generated TypeScript types: ${typesPath}`);
  }

  async run() {
    console.log('🚀 Starting image processing pipeline...');
    
    await this.ensureDirectories();
    const files = await this.getImageFiles();
    
    if (files.length === 0) {
      console.log('📁 No images found in raw directory.');
      console.log(`📁 Place images in: ${config.inputDir}`);
      return;
    }

    console.log(`📸 Found ${files.length} images to process`);
    
    for (const file of files) {
      const inputPath = path.join(config.inputDir, file);
      await this.processImage(inputPath, file);
    }

    await this.generateManifest();
    await this.generateTypeScriptTypes();
    
    console.log(`🎉 Processing complete! Processed ${this.processedImages.length} images`);
    console.log(`📊 Generated ${this.processedImages.length * Object.keys(config.sizes).length * config.formats.length} optimized images`);
  }
}

// Run the processor
const processor = new ImageProcessor();
processor.run().catch(console.error);

}

/**
 * Validate naming convention for a file
 */
function validateFile(filePath, fileName) {
  const errors = [];
  
  // Skip legacy files
  if (LEGACY_FILES.includes(filePath)) {
    return { isValid: true, errors: [], isLegacy: true };
  }
  
  // Check file extension and apply appropriate pattern
  if (fileName.endsWith('.tsx') || fileName.endsWith('.jsx')) {
    if (!PATTERNS.reactComponent.test(fileName)) {
      errors.push(`React component files should be PascalCase: ${fileName}`);
    }
  } else if (fileName.endsWith('.ts')) {
    if (!PATTERNS.typescriptFile.test(fileName)) {
      errors.push(`TypeScript files should be camelCase: ${fileName}`);
    }
  } else if (fileName.endsWith('.js')) {
    if (!PATTERNS.javascriptFile.test(fileName)) {
      errors.push(`JavaScript files should be camelCase: ${fileName}`);
    }
  } else if (fileName.includes('.config.')) {
    if (!PATTERNS.configFile.test(fileName)) {
      errors.push(`Configuration files should be kebab-case: ${fileName}`);
    }
  } else if (fileName.endsWith('.md') || fileName.endsWith('.txt')) {
    if (!PATTERNS.docFile.test(fileName)) {
      errors.push(`Documentation files should be PascalCase: ${fileName}`);
    }
  }
  
  return { isValid: errors.length === 0, errors, isLegacy: false };
}

/**
 * Validate naming convention for a directory
 */
function validateDirectory(dirPath, dirName) {
  const errors = [];
  
  // Skip legacy directories
  if (LEGACY_DIRS.includes(dirPath)) {
    return { isValid: true, errors: [], isLegacy: true };
  }
  
  // Skip root-level directories that have special naming
  if (dirPath === 'src' || dirPath === 'Q10UXPortfolio' || dirPath === 'chatgpt') {
    return { isValid: true, errors: [], isLegacy: false };
  }
  
  if (!PATTERNS.directory.test(dirName)) {
    errors.push(`Directories should be kebab-case: ${dirName}`);
  }
  
  return { isValid: errors.length === 0, errors, isLegacy: false };
}

/**
 * Recursively scan directory and validate naming conventions
 */
function scanDirectory(dirPath, relativePath = '') {
  const results = {
    files: [],
    directories: [],
    errors: [],
    legacyFiles: [],
    legacyDirectories: [],
  };
  
  try {
    const items = fs.readdirSync(dirPath);
    
    for (const item of items) {
      const fullPath = path.join(dirPath, item);
      const relativeItemPath = path.join(relativePath, item);
      const stats = fs.statSync(fullPath);
      
      if (shouldIgnore(item)) {
        continue;
      }
      
      if (stats.isDirectory()) {
        const validation = validateDirectory(relativeItemPath, item);
        results.directories.push({
          path: relativeItemPath,
          name: item,
          ...validation,
        });
        
        if (validation.isLegacy) {
          results.legacyDirectories.push(relativeItemPath);
        } else if (!validation.isValid) {
          results.errors.push(...validation.errors);
        }
        
        // Recursively scan subdirectories
        const subResults = scanDirectory(fullPath, relativeItemPath);
        results.files.push(...subResults.files);
        results.directories.push(...subResults.directories);
        results.errors.push(...subResults.errors);
        results.legacyFiles.push(...subResults.legacyFiles);
        results.legacyDirectories.push(...subResults.legacyDirectories);
      } else {
        const validation = validateFile(relativeItemPath, item);
        results.files.push({
          path: relativeItemPath,
          name: item,
          ...validation,
        });
        
        if (validation.isLegacy) {
          results.legacyFiles.push(relativeItemPath);
        } else if (!validation.isValid) {
          results.errors.push(...validation.errors);
        }
      }
    }
  } catch (error) {
    console.error(`Error scanning directory ${dirPath}:`, error.message);
  }
  
  return results;
}
